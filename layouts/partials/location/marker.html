{{/* Find the map marker page whose `site_information_page` refers to a location page. Takes the location page and returns the marker page. The marker owns the coordinates, so location pages read them from here rather than repeating them in their own front matter. */}} {{- $locationPage := . -}} {{- $markersSection := site.GetPage "locations" -}} {{- if not $markersSection -}} {{- errorf "the locations marker section was not found" -}} {{- end -}} {{- $marker := false -}} {{- range $candidate := $markersSection.Pages -}} {{- with $candidate.Params.site_information_page -}} {{- with site.GetPage . -}} {{- if eq .Path $locationPage.Path -}} {{- $marker = $candidate -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $marker -}} {{- errorf "location page %q has no marker in content/locations with a matching site_information_page" $locationPage.Path -}} {{- end -}} {{- return $marker -}}
